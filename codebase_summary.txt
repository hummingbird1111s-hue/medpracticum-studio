from django import forms
from django.contrib.auth.forms import AuthenticationForm, UserCreationForm

from .models import User

INPUT_CLASSES = (
    "w-full rounded-lg border border-gray-300 px-3 py-2 "
    "focus:outline-none focus:ring-2 focus:ring-emerald-500"
)


class StyledFormMixin:
    """Adds Tailwind classes to every field."""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        for field in self.fields.values():
            field.widget.attrs.setdefault('class', INPUT_CLASSES)


class RegistrationForm(StyledFormMixin, UserCreationForm):
    level = forms.TypedChoiceField(
        choices=[(100, 'Level 100'), (200, 'Level 200'), (300, 'Level 300'), (400, 'Level 400')],
        coerce=int,
        initial=300,
    )

    class Meta(UserCreationForm.Meta):
        model = User
        fields = (
            'first_name', 'last_name', 'username', 'email',
            'index_number', 'department', 'level', 'phone_number',
        )

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.fields['first_name'].required = True
        self.fields['last_name'].required = True


class LoginForm(StyledFormMixin, AuthenticationForm):
    pass

from django.contrib.auth.models import AbstractUser
from django.db import models


class User(AbstractUser):
    DEPARTMENT_CHOICES = [
        ('HIM', 'Health Information Management'),
        ('PA', 'Physician Assistantship'),
        ('MED_LAB', 'Medical Laboratory Sciences'),
        ('DISEASE_CTRL', 'Disease Control & Epidemiology'),
        ('MENTAL_HEALTH', 'Community Mental Health'),
        ('NUTRITION', 'Community Health Nutrition'),
        ('HEALTH_PROM', 'Health Promotion'),
    ]

    index_number = models.CharField(max_length=20, unique=True)
    department = models.CharField(max_length=20, choices=DEPARTMENT_CHOICES)
    level = models.IntegerField(default=300)  # 100, 200, 300, 400

        return f"{self.get_full_name() or self.username} ({self.index_number})"

from django.contrib.auth import login
from django.shortcuts import redirect, render

from .forms import RegistrationForm


def register(request):
    if request.user.is_authenticated:
        return redirect('dashboard')

    if request.method == 'POST':
        form = RegistrationForm(request.POST)
        if form.is_valid():
            user = form.save()
            login(request, user)
            return redirect('dashboard')
    else:
        form = RegistrationForm()

    return render(request, 'accounts/register.html', {'form': form})

import random
from datetime import timedelta

from django.core.management.base import BaseCommand, CommandError

from apps.accounts.models import User
from apps.logs.department_config import get_department_config
from apps.logs.models import DailyLog

ACTIVITIES = {
    'HIM': [
        "Registered new and returning patients at the OPD records desk and created new folders.",
        "Retrieved and refiled patient folders for the consulting rooms throughout the morning.",
        "Assisted with ICD-10 coding of discharge summaries under the supervision of the coding officer.",
        "Traced and resolved misplaced folders by cross-checking the folder movement register.",
        "Compiled daily OPD attendance figures and entered them into the electronic health record system.",
        "Helped prepare NHIS claims batches and checked them for missing diagnosis codes.",
    ],
    'PA': [
        "Observed and assisted with history taking for patients presenting at the OPD.",
        "Performed supervised physical examinations and discussed findings with the physician assistant.",
        "Took vital signs and assisted with triage of patients in the emergency unit.",
        "Reviewed common presentations such as fever, cough and abdominal pain with the clinical team.",
        "Practised writing SOAP notes and discussed differential diagnoses with the supervisor.",
        "Assisted with dressing and minor procedures under supervision.",
    ],
}

CHALLENGES = [
    ("Long queues at peak hours slowed down service delivery.",
     "Worked with staff to prioritise urgent cases and organise patients by arrival time."),
    ("Some patient records were incomplete when retrieved.",
     "Cross-checked with the registers and asked staff to help complete the missing details."),
    ("Difficulty recalling unfamiliar terminology during the first weeks.",
     "Kept a personal glossary and asked the supervisor to explain terms after each shift."),
]


class Command(BaseCommand):
    help = "Create sample daily logs for a user's latest attachment (for testing reports)."

    def add_arguments(self, parser):
        parser.add_argument('username')
        parser.add_argument('--days', type=int, default=20)
        parser.add_argument('--clear', action='store_true', help='Delete previously created sample logs')

    def handle(self, *args, **opts):
        try:
            user = User.objects.get(username=opts['username'])
        except User.DoesNotExist:
            raise CommandError('No user with that username.')

        session = user.attachments.order_by('-start_date', '-id').first()
        if session is None:
            raise CommandError('That user has no attachment yet.')

        if opts['clear']:
            removed = 0
            for log in session.daily_logs.all():
                if '#SAMPLE' in (log.tags or []):
                    log.delete()
                    removed += 1
            self.stdout.write(f'Removed {removed} sample logs.')
            return

        dept = get_department_config(user.department)
        rng = random.Random(42)
        activities = ACTIVITIES.get(user.department)
        created = 0
        day = session.start_date

        while created < opts['days'] and day <= session.end_date:
            if day.weekday() < 5 and not session.daily_logs.filter(date=day).exists():
                week = (day - session.start_date).days // 7
                unit = dept['units'][week % len(dept['units'])]
                summary = rng.choice(activities) if activities else (
                    f"Worked in {unit}, assisting staff with routine tasks and observing procedures."
                )
                challenge, solution = ('', '')
                if created % 4 == 3:
                    challenge, solution = CHALLENGES[(created // 4) % len(CHALLENGES)]

                DailyLog.objects.create(
                    attachment=session,
                    date=day,
                    week_number=week + 1,
                    unit_sub_department=unit,
                    activity_summary=summary,
                    tags=rng.sample(dept['tags'], k=min(2, len(dept['tags']))) + ['#SAMPLE'],
                    numeric_tallies={t['key']: rng.randint(5, 60) for t in dept['tallies']},
                    department_details={d['key']: 'Sample entry (test data)' for d in dept['details']},
                    skills_acquired='Improved accuracy, speed and confidence with routine tasks.' if created % 3 == 0 else '',
                    challenges_faced=challenge,
                    solutions_applied=solution,
                )
                created += 1
            day += timedelta(days=1)

        self.stdout.write(self.style.SUCCESS(f'Created {created} sample logs.'))

    verification_token = models.UUIDField(default=uuid.uuid4, editable=False, unique=True)
    is_completed = models.BooleanField(default=False)

    def clean(self):
        if self.start_date and self.end_date and self.end_date < self.start_date:
            raise ValidationError({'end_date': 'End date cannot be before the start date.'})

    def __str__(self):
        return f"{self.user} @ {self.facility} ({self.academic_year})"


class DailyLog(models.Model):
    attachment = models.ForeignKey(AttachmentSession, on_delete=models.CASCADE, related_name='daily_logs')
    date = models.DateField()
    week_number = models.IntegerField()  # Weeks 1 through 8
    unit_sub_department = models.CharField(max_length=100)  # e.g., Records Unit, OPD, Lab Section

    # Content
    activity_summary = models.TextField(help_text="Detailed description of tasks completed")
    tags = models.JSONField(default=list, blank=True)  # e.g., ["#ICD10", "#Registration"]

    # Quantitative
    numeric_tallies = models.JSONField(default=dict, blank=True)  # e.g., {"patients_registered": 45}

    # Department-specific text answers (e.g., chief complaints, EHR system used)
    department_details = models.JSONField(default=dict, blank=True)

    # Qualitative
    skills_acquired = models.TextField(blank=True)
    challenges_faced = models.TextField(blank=True)
    solutions_applied = models.TextField(blank=True)

    # Status
    created_offline = models.BooleanField(default=False)
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ['date']
        unique_together = ['attachment', 'date']

    def save(self, *args, **kwargs):
        # Empty form/API input arrives as None; store sensible defaults instead
        if self.tags is None:
            self.tags = []
        if self.numeric_tallies is None:
            self.numeric_tallies = {}
        if self.department_details is None:
            self.department_details = {}
        super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.date} - {self.unit_sub_department}"

def _to_int(value):
    try:
        return int(value)
    except (TypeError, ValueError):
        return None


def _clean_tags(raw):
    if not isinstance(raw, list):
        return []
    cleaned = []
    for tag in raw:
        text = str(tag).strip()[:40]
        if not text:
            continue
        if not text.startswith('#'):
            text = '#' + text
        if text not in cleaned:
            cleaned.append(text)
    return cleaned[:MAX_TAGS]


def _clean_tallies(raw):
    if not isinstance(raw, dict):
        return {}
    cleaned = {}
    for key, value in raw.items():
        try:
            number = float(value)
        except (TypeError, ValueError):
            continue
        if number < 0:
            continue
        cleaned[str(key)[:50]] = int(number) if number.is_integer() else number
    return cleaned


def _clean_details(raw):
    if not isinstance(raw, dict):
        return {}
    cleaned = {}
    for key, value in raw.items():
        text = str(value).strip()[:2000]
        if text:
            cleaned[str(key)[:50]] = text
    return cleaned


def _save_item(sessions, item):
    session = sessions.get(_to_int(item.get('attachment_id')))
    if session is None:
        raise ValueError('Unknown attachment session.')

    try:
        log_date = parse_date(str(item.get('date', '')))
    except ValueError:
        log_date = None
    if log_date is None:
        raise ValueError('Invalid date.')
    if not (session.start_date <= log_date <= session.end_date):
        raise ValueError('That date is outside your attachment period.')

    unit = str(item.get('unit_sub_department', '')).strip()[:100]
    summary = str(item.get('activity_summary', '')).strip()
    if not unit:
        raise ValueError('Unit is required.')
    if not summary:
        raise ValueError('Activity summary is required.')

    DailyLog.objects.update_or_create(
        attachment=session,
        date=log_date,
        defaults={
            'week_number': (log_date - session.start_date).days // 7 + 1,
            'unit_sub_department': unit,
            'activity_summary': summary,
            'tags': _clean_tags(item.get('tags')),
            'numeric_tallies': _clean_tallies(item.get('numeric_tallies')),
            'department_details': _clean_details(item.get('department_details')),
            'skills_acquired': str(item.get('skills_acquired', '')).strip(),
            'challenges_faced': str(item.get('challenges_faced', '')).strip(),
            'solutions_applied': str(item.get('solutions_applied', '')).strip(),
            'created_offline': bool(item.get('created_offline')),
        },
    )


@require_POST
def sync_logs(request):
    if not request.user.is_authenticated:
        return JsonResponse({'error': 'authentication required'}, status=401)

    try:
        items = json.loads(request.body)['logs']
        if not isinstance(items, list):
            raise ValueError
    except (ValueError, KeyError, TypeError):
        return JsonResponse({'error': 'invalid payload'}, status=400)

    sessions = {s.id: s for s in request.user.attachments.all()}
    synced, errors = [], []

    for item in items[:MAX_LOGS_PER_SYNC]:
        if not isinstance(item, dict):
            errors.append({'key': '', 'error': 'Invalid log entry.'})
            continue
        key = str(item.get('key', ''))
        try:
            _save_item(sessions, item)
            synced.append(key)
        except ValueError as exc:
            errors.append({'key': key, 'error': str(exc)})

    return JsonResponse({'synced': synced, 'errors': errors})

def _to_data_uri(fig):
    buf = io.BytesIO()
    fig.savefig(buf, format='png', dpi=200, bbox_inches='tight')
    plt.close(fig)
    return 'data:image/png;base64,' + base64.b64encode(buf.getvalue()).decode('ascii')


def _wrap(labels, width=22):
    return [textwrap.fill(str(label), width) for label in labels]


def bar_chart(labels, values, value_label='', horizontal=False):
    labels = _wrap(labels)
    if horizontal:
        fig, ax = plt.subplots(figsize=(6.2, max(2.8, 0.5 * len(labels) + 1)))
        bars = ax.barh(labels, values, color=GREEN)
        ax.invert_yaxis()
        ax.set_xlabel(value_label)
    else:
        fig, ax = plt.subplots(figsize=(6.2, 3.6))
        bars = ax.bar(labels, values, color=GREEN)
        ax.set_ylabel(value_label)
        plt.setp(ax.get_xticklabels(), rotation=30, ha='right')
    ax.bar_label(bars, padding=2)
    return _to_data_uri(fig)


def line_chart(labels, values, value_label=''):
    fig, ax = plt.subplots(figsize=(6.2, 3.4))
    ax.plot(labels, values, marker='o', color=GREEN, linewidth=2)
    for x, y in zip(labels, values):
        ax.annotate(f'{y:g}', (x, y), textcoords='offset points', xytext=(0, 6), ha='center', fontsize=9)
    ax.set_ylabel(value_label)
    ax.set_ylim(bottom=0)
    return _to_data_uri(fig)

def _num(value):
    try:
        number = float(value)
    except (TypeError, ValueError):
        return value
    return int(number) if number.is_integer() else round(number, 1)


def _plural(n, word):
    return f"{n} {word}" if n == 1 else f"{n} {word}s"


def _paragraphs(text):
    return [p.strip() for p in re.split(r'\n\s*\n', text or '') if p.strip()]


def _lines(text):
    return [line.strip() for line in (text or '').splitlines() if line.strip()]


def _first_sentence(text, limit=140):
    text = ' '.join((text or '').split())
    cut = re.split(r'(?<=[.!?])\s', text, maxsplit=1)[0]
    return cut if len(cut) <= limit else cut[:limit].rstrip() + '...'

    def _weeks(self):
        detail_labels = {d['key']: d['label'] for d in self.dept['details']}
        grouped = OrderedDict()
        for log in self.logs:
            grouped.setdefault(log.week_number, []).append(log)

        weeks = []
        for number, items in grouped.items():
            units = sorted({log.unit_sub_department for log in items})
            sentences, seen = [], set()
            for log in items:
                sentence = _first_sentence(log.activity_summary).rstrip('.')
                if sentence and sentence not in seen:
                    seen.add(sentence)
                    sentences.append(sentence)
                if len(sentences) == 3:
                    break
            digest = ('Main activities included: ' + '; '.join(sentences) + '.') if sentences else ''
            weeks.append({
                'number': number,
                'start': items[0].date,
                'end': items[-1].date,
                'days': len(items),
                'units': units,
                'digest': digest,
                'logs': [{
                    'date': log.date,
                    'unit': log.unit_sub_department,
                    'summary': log.activity_summary,
                    'details': [
                        (detail_labels.get(k, k.replace('_', ' ').capitalize()), v)
                        for k, v in (log.department_details or {}).items()
                    ],
                } for log in items],
            })

    def _tally_stats(self):
        labels = {t['key']: t['label'] for t in self.dept['tallies']}

        keys = [t['key'] for t in self.dept['tallies'] if totals.get(t['key'])]

    def build_pdf(self):
        opt = self.options
        target = opt.get('target_pages')
        spacing = opt['line_spacing']

        document, pages, dropped = self._document(spacing, 0)
        chosen = {'doc': document, 'pages': pages, 'spacing': spacing, 'level': 0, 'dropped': dropped}
        grow = bool(target) and pages < target

        if target and pages != target:
            if grow:
                steps = [(sp, 0) for sp in FIT_SPACINGS if sp > spacing]
            else:
                steps = [(sp, 0) for sp in sorted((x for x in FIT_SPACINGS if x < spacing), reverse=True)]
                steps += [(min(FIT_SPACINGS), lvl) for lvl in range(1, MAX_CONDENSE_LEVEL + 1)]

            for sp, lvl in steps:
                doc, n, drp = self._document(sp, lvl)
                candidate = {'doc': doc, 'pages': n, 'spacing': sp, 'level': lvl, 'dropped': drp}
                if grow:
                    if abs(n - target) < abs(chosen['pages'] - target):
                        chosen = candidate
                    if n >= target:
                        break
                else:
                    if n <= target:
                        chosen = candidate
                        break
                    if n < chosen['pages']:
                        chosen = candidate

        final_pages = chosen['pages']
        notes = []
        if target:
            if final_pages == target:
                notes.append(f'Exactly {target} pages, as you asked.')
            elif grow:
                notes.append(f'Your logs fill only {final_pages} pages even at the widest spacing, short of your '
                             f'{target}-page target. Add more detail to your daily logs, or switch on Appendix A '
                             'and the charts, to reach it.')
            elif final_pages > target:
                notes.append(f'The smallest we could make it is {final_pages} pages, above your {target}-page '
                             'target. Try a smaller font size or narrower margins.')
            else:
                notes.append(f'The closest fit was {final_pages} pages (target {target}).')
            if chosen['dropped']:
                notes.append('To fit, we changed: ' + ', '.join(chosen['dropped']) + '.')
        if final_pages < MIN_PAGES_GUIDELINE:
            notes.append(f'The COHY guideline is at least {MIN_PAGES_GUIDELINE} pages.')

        self.result = {
            'pages': final_pages, 'spacing': chosen['spacing'], 'level': chosen['level'],
            'dropped': chosen['dropped'], 'target': target, 'note': ' '.join(notes),
        }
        return chosen['doc'].write_pdf()

    def save(self):

        }

        report.pdf_file.save(filename, ContentFile(pdf_bytes), save=True)
        return report

def normalize_options(raw):

    raw = raw or {}

import logging
import os

from django.conf import settings
from django.contrib import messages
from django.contrib.auth.decorators import login_required
from django.http import FileResponse, Http404, HttpResponse
from django.shortcuts import redirect, render
from django.views.decorators.http import require_POST

from .forms import LAYOUT_FIELDS, SECTION_FIELDS, ReportContentForm, ReportOptionsForm
from .models import GeneratedReport, SchoolProfile
from .services import PracticumReportBuilder
from .standards import MIN_PAGES_GUIDELINE, SCHOOL_STANDARD, normalize_options

logger = logging.getLogger(__name__)


def _latest_session(user):
    return user.attachments.order_by('-start_date', '-id').first()


def _build(request, session):
    builder = PracticumReportBuilder(session)
    try:
        builder.save()
    except Exception as exc:
        logger.exception('Report generation failed')
        messages.error(request, f'Could not build the report: {exc}')
        return False

    result = builder.result
    text = f"Your report is ready: {result['pages']} pages."
    if result.get('note'):
        text += ' ' + result['note']
    messages.success(request, text)
    return True


def _session_with_logs(request):
    session = _latest_session(request.user)
    if session is None:
        messages.error(request, 'Set up your attachment first.')
        return None, redirect('start_attachment')
    if not session.daily_logs.exists():
        messages.error(request, 'Add at least one daily log before working on a report.')
        return None, redirect('dashboard')
    return session, None


@login_required
@require_POST
def generate_report(request):
    """Rebuild using the student's saved settings (or the official format if none)."""
    session, bounce = _session_with_logs(request)
    if bounce:
        return bounce
    _build(request, session)
    return redirect('dashboard')


@login_required
def report_options(request):
    session, bounce = _session_with_logs(request)
    if bounce:
        return bounce

    allow_custom = getattr(settings, 'REPORT_ALLOW_CUSTOM_FORMAT', True)
    report, _ = GeneratedReport.objects.get_or_create(attachment=session)
    form = None

    if request.method == 'POST':
        if request.POST.get('action') == 'official' or not allow_custom:
            report.options = normalize_options({})
            report.save(update_fields=['options'])
            _build(request, session)
            return redirect('dashboard')

        form = ReportOptionsForm(request.POST)
        if form.is_valid():
            report.options = normalize_options(form.cleaned_data)
            report.save(update_fields=['options'])
            _build(request, session)
            return redirect('dashboard')
    elif allow_custom:
        form = ReportOptionsForm(initial=normalize_options(report.options))

    school = SchoolProfile.objects.first()
    return render(request, 'reports/options.html', {
        'form': form,
        'layout_fields': [form[n] for n in LAYOUT_FIELDS] if form else [],
        'section_fields': [form[n] for n in SECTION_FIELDS] if form else [],
        'standard': SCHOOL_STANDARD,
        'min_pages': MIN_PAGES_GUIDELINE,
        'allow_custom': allow_custom,
        'has_logo': bool(school and school.logo),
    })


@login_required
def report_content(request):
    """Lets the student edit the automatic text in their own words."""
    session, bounce = _session_with_logs(request)
    if bounce:
        return bounce

    report, _ = GeneratedReport.objects.get_or_create(attachment=session)
    builder = PracticumReportBuilder(session)
    defaults = builder.content_defaults()

    if request.method == 'POST':
        if request.POST.get('action') == 'reset':
            report.content = {}
            report.save(update_fields=['content'])
            messages.success(request, 'Your report text was reset to the automatic draft.')
            return redirect('report_content')

        form = ReportContentForm(request.POST)
        if form.is_valid():
            # Keep only what the student actually changed, so untouched text stays up to date
            report.content = {
                key: value.strip()
                for key, value in form.cleaned_data.items()
                if value.strip() and value.strip() != defaults.get(key, '').strip()
            }
            report.save(update_fields=['content'])
            messages.success(request, 'Saved. Generate your report to see the changes.')
            return redirect('report_options')
    else:
        form = ReportContentForm(initial=builder.effective_content())

    return render(request, 'reports/content.html', {'form': form})


def _pdf_response(request, inline):
    session = _latest_session(request.user)
    report = getattr(session, 'report', None) if session else None
    if not report or not report.pdf_file:
        raise Http404('No report has been generated yet.')
    return FileResponse(
        report.pdf_file.open('rb'),
        as_attachment=not inline,
        filename=os.path.basename(report.pdf_file.name),
        content_type='application/pdf',
    )


@login_required
def download_report(request):
    return _pdf_response(request, inline=False)


@login_required
def view_report(request):
    """Opens the PDF in the browser so it can be printed."""
    return _pdf_response(request, inline=True)


@login_required
def preview_report(request):
    """Fast HTML preview (no PDF) for checking wording and charts."""
    session = _latest_session(request.user)
    if session is None:
        raise Http404
    return HttpResponse(PracticumReportBuilder(session).render_html())

OUT = Path('static/icons')
OUT.mkdir(parents=True, exist_ok=True)

GREEN = (5, 150, 105)
WHITE = (255, 255, 255)


def make_icon(size, filename, text_ratio):
    img = Image.new('RGB', (size, size), GREEN)
    draw = ImageDraw.Draw(img)
    font = ImageFont.load_default(size=int(size * text_ratio))
    text = 'MP'
    left, top, right, bottom = draw.textbbox((0, 0), text, font=font)
    x = (size - (right - left)) / 2 - left
    y = (size - (bottom - top)) / 2 - top
    draw.text((x, y), text, font=font, fill=WHITE)
    img.save(OUT / filename)
    print('created', OUT / filename)

document.addEventListener('alpine:init', () => {
  Alpine.data('logForm', (cfg) => {
    let recognition = null; // kept outside Alpine so the browser object isn't proxied

    return {
      cfg,
      form: {
        date: '',
        unit_sub_department: '',
        activity_summary: '',
        skills_acquired: '',
        challenges_faced: '',
        solutions_applied: '',
        tags: [],
        tallies: {},
        details: {},
      },
      tagInput: '',
      listening: null,
      speechSupported: !!(window.SpeechRecognition || window.webkitSpeechRecognition),
      saving: false,
      message: '',
      messageType: 'info',

      init() {
        cfg.tallies.forEach((t) => { this.form.tallies[t.key] = ''; });
        cfg.details.forEach((d) => { this.form.details[d.key] = ''; });

        const now = new Date();
        const localToday = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0') + '-' + String(now.getDate()).padStart(2, '0');
        let date = cfg.existing ? cfg.existing.date : localToday;
        if (date < cfg.start_date) date = cfg.start_date;
        if (date > cfg.end_date) date = cfg.end_date;
        this.form.date = date;

        const ex = cfg.existing;
        if (ex) {
          this.form.unit_sub_department = ex.unit_sub_department || '';
          ['activity_summary', 'skills_acquired', 'challenges_faced', 'solutions_applied']
            .forEach((f) => { this.form[f] = ex[f] || ''; });
          this.form.tags = [...(ex.tags || [])];
          Object.entries(ex.tallies || {}).forEach(([k, v]) => { this.form.tallies[k] = v; });
          Object.entries(ex.details || {}).forEach(([k, v]) => { this.form.details[k] = v; });
        }
      },

      get weekNumber() {
        if (!this.form.date) return '';
        const d = new Date(this.form.date + 'T00:00:00');
        const s = new Date(cfg.start_date + 'T00:00:00');
        const days = Math.round((d - s) / 86400000);
        return Math.floor(days / 7) + 1;
      },

      // ----- Tags -----
      addTag(value) {
        let tag = (value !== undefined ? value : this.tagInput).trim();
        if (!tag) return;
        if (!tag.startsWith('#')) tag = '#' + tag;
        if (!this.form.tags.includes(tag) && this.form.tags.length < 20) this.form.tags.push(tag);
        this.tagInput = '';
      },
      removeTag(tag) {
        this.form.tags = this.form.tags.filter((t) => t !== tag);
      },

      // ----- Voice dictation -----
      dictate(field) {
        if (!this.speechSupported) return;
        if (this.listening === field && recognition) {
          recognition.stop();
          return;
        }
        if (recognition) recognition.stop();

        const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
        const rec = new SR();
        rec.lang = 'en-US'; // try 'en-GH' if it recognises your accent better
        rec.continuous = true;
        rec.interimResults = false;

        rec.onstart = () => { this.listening = field; };
        rec.onresult = (e) => {
          for (let i = e.resultIndex; i < e.results.length; i++) {
            if (e.results[i].isFinal) {
              this.form[field] = (this.form[field] + ' ' + e.results[i][0].transcript).trim();
            }
          }
        };
        rec.onend = () => { this.listening = null; };
        rec.onerror = (e) => {
          this.listening = null;
          if (e.error === 'not-allowed') this.setMessage('Microphone permission was blocked.', 'error');
          else if (e.error === 'network') this.setMessage('Dictation needs an internet connection.', 'error');
        };

        recognition = rec;
        rec.start();
      },

      // ----- Saving -----
      validate() {
        if (!this.form.date) return 'Pick a date.';
        if (this.form.date < cfg.start_date || this.form.date > cfg.end_date) {
          return 'That date is outside your attachment period (' + cfg.start_date + ' to ' + cfg.end_date + ').';
        }
        if (!this.form.unit_sub_department.trim()) return 'Choose or type the unit you worked in.';
        if (!this.form.activity_summary.trim()) return 'Describe what you did today.';
        return '';
      },

      buildPayload() {
        const tallies = {};
        Object.entries(this.form.tallies).forEach(([k, v]) => {
          if (v !== '' && v !== null && !Number.isNaN(Number(v))) tallies[k] = Number(v);
        });
        const details = {};
        Object.entries(this.form.details).forEach(([k, v]) => {
          if (String(v || '').trim()) details[k] = String(v).trim();
        });
        return {
          attachment_id: cfg.attachment_id,
          date: this.form.date,
          unit_sub_department: this.form.unit_sub_department.trim(),
          activity_summary: this.form.activity_summary.trim(),
          skills_acquired: this.form.skills_acquired.trim(),
          challenges_faced: this.form.challenges_faced.trim(),
          solutions_applied: this.form.solutions_applied.trim(),
          tags: [...this.form.tags],
          numeric_tallies: tallies,
          department_details: details,
        };
      },

      async queueOffline(payload, authProblem) {
        await window.MPOffline.queueLog(payload);
        this.setMessage(
          authProblem
            ? 'Session expired. Saved on this device; log in again to sync.'
            : 'Saved on this device. It will sync automatically when you are back online.',
          'success'
        );
      },

      async submit() {
        const problem = this.validate();
        if (problem) { this.setMessage(problem, 'error'); return; }

        const payload = this.buildPayload();
        this.saving = true;
        try {
          if (!navigator.onLine) {
            await this.queueOffline(payload, false);
            return;
          }

          let result;
          try {
            result = await window.MPOffline.sendLogs([payload]);
          } catch (err) {
            await this.queueOffline(payload, err.code === 'auth');
            return;
          }

          if (result.errors && result.errors.length) {
            this.setMessage(result.errors[0].error, 'error');
            return;
          }
          this.setMessage('Saved ✓', 'success');
          setTimeout(() => { window.location.href = cfg.dashboard_url; }, 700);
        } catch (e) {
          this.setMessage('Could not save on this device: ' + (e && e.message ? e.message : e), 'error');
        } finally {
          this.saving = false;
        }
      },

      setMessage(text, type) {
        this.message = text;
        this.messageType = type;
      },
    };
  });
});

// IndexedDB queue + auto-sync. Loaded on every page via base.html.
(function () {
  const DB_NAME = 'medpracticum';
  const STORE = 'offline_logs';
  let syncing = false;
  let lastError = '';

  function openDB() {
    return new Promise((resolve, reject) => {
      if (!window.indexedDB) {
        reject(new Error('This browser cannot store data on the device.'));
        return;
      }
      const req = indexedDB.open(DB_NAME, 1);
      req.onupgradeneeded = () => req.result.createObjectStore(STORE, { keyPath: 'key' });
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => reject(req.error);
    });
  }

  async function tx(mode, fn) {
    const db = await openDB();
    return new Promise((resolve, reject) => {
      const t = db.transaction(STORE, mode);
      const req = fn(t.objectStore(STORE));
      t.oncomplete = () => resolve(req ? req.result : undefined);
      t.onerror = () => reject(t.error);
      t.onabort = () => reject(t.error);
    });
  }

  function getCookie(name) {
    const match = document.cookie.split('; ').find((c) => c.startsWith(name + '='));
    return match ? decodeURIComponent(match.split('=')[1]) : '';
  }

  async function sendLogs(logs) {
    const res = await fetch('/api/logs/sync/', {
      method: 'POST',
      credentials: 'same-origin',
      headers: {
        'Content-Type': 'application/json',
        'X-CSRFToken': getCookie('csrftoken'),
      },
      body: JSON.stringify({ logs }),
    });
    if (res.status === 401 || res.status === 403) {
      const err = new Error('Please log in again.');
      err.code = 'auth';
      throw err;
    }
    return res.json();
  }

  async function queueLog(original) {
    // Copy into plain data first: Alpine's proxy objects can't be stored in IndexedDB
    const payload = JSON.parse(JSON.stringify(original));
    payload.key = payload.attachment_id + '-' + payload.date; // same day = overwrite
    payload.created_offline = true;
    await tx('readwrite', (s) => s.put(payload));
    updateBanner();
  }

  function pendingCount() {
    return tx('readonly', (s) => s.count()).catch(() => 0);
  }

  async function updateBanner() {
    const el = document.getElementById('mp-banner');
    if (!el) return;
    const n = await pendingCount();
    const offline = !navigator.onLine;
    if (!offline && n === 0) {
      el.classList.add('hidden');
      return;
    }
    const waiting = n + ' log' + (n === 1 ? '' : 's') + ' waiting to sync';
    let text = offline
      ? "You're offline" + (n ? ' · ' + waiting : ' · new logs will be saved on this device')
      : waiting + '…';
    if (lastError) text += ' (' + lastError + ')';
    el.textContent = text;
    el.classList.remove('hidden');
  }

  async function syncQueue() {
    if (syncing || !navigator.onLine) return;
    syncing = true;
    try {
      const logs = await tx('readonly', (s) => s.getAll());
      if (!logs || !logs.length) return;
      const result = await sendLogs(logs);
      for (const key of result.synced || []) {
        await tx('readwrite', (s) => s.delete(key));
      }
      lastError = result.errors && result.errors.length ? result.errors[0].error : '';
      if ((result.synced || []).length) {
        window.dispatchEvent(new CustomEvent('mp:synced', { detail: result }));
      }
    } catch (e) {
      // Stay queued; we'll try again next time.
    } finally {
      syncing = false;
      updateBanner();
    }
  }

  window.addEventListener('online', () => { updateBanner(); syncQueue(); });
  window.addEventListener('offline', updateBanner);
  document.addEventListener('DOMContentLoaded', () => { updateBanner(); syncQueue(); });

  window.MPOffline = { queueLog, syncQueue, sendLogs, pendingCount };
})();

(function () {
  // 1. Register the service worker
  if ('serviceWorker' in navigator) {
    window.addEventListener('load', () => {
      navigator.serviceWorker.register('/service-worker.js', { scope: '/' }).catch(() => {});
    });
  }

  // 2. "Install app" button (Chrome/Edge/Android). iPhones use Share → Add to Home Screen.
  let deferredPrompt = null;
  const btn = document.getElementById('install-btn');

  window.addEventListener('beforeinstallprompt', (e) => {
    e.preventDefault();
    deferredPrompt = e;
    if (btn) btn.classList.remove('hidden');
  });

  if (btn) {
    btn.addEventListener('click', async () => {
      if (!deferredPrompt) return;
      deferredPrompt.prompt();
      await deferredPrompt.userChoice;
      deferredPrompt = null;
      btn.classList.add('hidden');
    });
  }

  window.addEventListener('appinstalled', () => {
    if (btn) btn.classList.add('hidden');
  });

  // 3. On logout, wipe saved pages so the next person on a shared phone can't see them
  document.querySelectorAll('form[data-logout]').forEach((form) => {
    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      try {
        const keys = await caches.keys();
        await Promise.all(keys.filter((k) => k.startsWith('mp-pages-')).map((k) => caches.delete(k)));
      } catch (err) { /* ignore */ }
      form.submit();
    });
  });
})();

{% load static %}
const VERSION = 'v2'; // bump this number when you want everyone to drop old caches
const STATIC_CACHE = 'mp-static-' + VERSION;
const PAGES_CACHE = 'mp-pages-' + VERSION;
const OFFLINE_URL = '/offline/';
const CACHED_PAGES = ['/', '/logs/new/']; // pages students need with no signal

const PRECACHE = [
  OFFLINE_URL,
  '{% static "css/tailwind.css" %}',
  '{% static "vendor/alpine.min.js" %}',
  '{% static "vendor/htmx.min.js" %}',
  '{% static "js/offline-sync.js" %}',
  '{% static "js/log_form.js" %}',
  '{% static "js/pwa.js" %}',
  '{% static "icons/icon-192.png" %}',
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(STATIC_CACHE)
      .then((cache) => cache.addAll(PRECACHE))
      .then(() => self.skipWaiting())
  );
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys()
      .then((keys) => Promise.all(
        keys
          .filter((k) => k.startsWith('mp-') && k !== STATIC_CACHE && k !== PAGES_CACHE)
          .map((k) => caches.delete(k))
      ))
      .then(() => self.clients.claim())
  );
});

self.addEventListener('fetch', (event) => {
  const req = event.request;
  if (req.method !== 'GET') return;

  const url = new URL(req.url);
  if (url.origin !== self.location.origin) return;
  if (url.pathname.startsWith('/api/') || url.pathname.startsWith('/admin/')) return;

  if (url.pathname.startsWith('/static/')) {
    event.respondWith(staleWhileRevalidate(req));
    return;
  }

  if (req.mode === 'navigate') {
    event.respondWith(networkFirstPage(req, url));
  }
});

// Static files: answer instantly from cache, refresh the cache in the background.
async function staleWhileRevalidate(req) {
  const cache = await caches.open(STATIC_CACHE);
  const cached = await cache.match(req);
  const network = fetch(req)
    .then((res) => {
      if (res.ok) cache.put(req, res.clone());
      return res;
    })
    .catch(() => null);
  return cached || (await network) || Response.error();
}

// Pages: try the server first; if it fails, fall back to the last saved copy.
async function networkFirstPage(req, url) {
  const cache = await caches.open(PAGES_CACHE);
  try {
    const res = await fetch(req);
    // Never store redirects (e.g. bounced to the login page)
    if (res.ok && !res.redirected && CACHED_PAGES.includes(url.pathname)) {
      cache.put(url.pathname, res.clone());
    }
    return res;
  } catch (err) {
    const cached = await cache.match(url.pathname);
    if (cached) return cached;
    const offline = await caches.match(OFFLINE_URL);
    return offline || Response.error();
  }
}
